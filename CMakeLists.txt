cmake_minimum_required(VERSION 3.25)
project(cs2esp LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This project requires 64-bit Windows and MSVC.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(/utf-8 /permissive- /EHsc /MP /W3)
add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
set(GETREKT_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(GETREKT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(DrateCoreD)
set(OC_EMBEDDED ON CACHE BOOL "" FORCE)
set(OPENCHANGER_DEV ON CACHE BOOL "" FORCE)
add_subdirectory(OpenChangerCore EXCLUDE_FROM_ALL)
add_library(cs2esp_core STATIC Global.cpp SDK.cpp Hooks.cpp ESP.cpp PlayerSnapshot.cpp
    Aim.cpp AimChecks.cpp Backtrack.cpp LegitBot.cpp Misc.cpp NativeJump.cpp ThirdPerson.cpp ExtraRuntime.cpp ClientConVars.cpp Config.cpp Chams.cpp WorldVisuals.cpp SkinChanger.cpp SkinInventoryBridge.cpp ManualMouse.cpp MousePollGate.cpp
    DrateCoreD/vendor/imgui/backends/imgui_impl_dx11.cpp
    DrateCoreD/vendor/imgui/backends/imgui_impl_win32.cpp)
target_include_directories(cs2esp_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} DrateCoreD/vendor/imgui/backends)
target_compile_features(cs2esp_core PRIVATE cxx_std_23)
target_link_libraries(cs2esp_core PUBLIC getrekt_ui oc_objects d3d11 dxgi d3dcompiler dwmapi winmm)
add_library(cs2esp SHARED DLLMain.cpp)
target_sources(cs2esp PRIVATE $<TARGET_OBJECTS:oc_objects>)
target_link_libraries(cs2esp PRIVATE cs2esp_core)
set_target_properties(cs2esp PROPERTIES PREFIX "" OUTPUT_NAME "cs2esp")
